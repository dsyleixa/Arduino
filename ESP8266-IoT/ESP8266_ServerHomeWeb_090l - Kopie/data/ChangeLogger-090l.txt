// ===================================================================
// ChangeLogger() 090k: prüft Änderungen aller OUT- und c*out-Werte
//      und loggt welche Änderungen erfolgt sind.
//      Zusätzlich wird die vorherige und vor-vorherige
//      an Clients gesendete Message angezeigt.
// ===================================================================

void ChangeLogger() {
   // Hilfs-Lambda für Wertevergleich + Update
   bool changed = false;

   auto check = [&](const char* name, int8_t &oldVal, int8_t newVal) {
      if (oldVal != newVal) {
         SwitchChangeLog = (String)"[CHANGE] " + name + ": " + oldVal + " -> " + newVal;
         Serial.println(SwitchChangeLog);
         oldVal = newVal;
         changed = true;
      }
   };

   // Alle OUT-Werte prüfen
   check("OUT1", old_OUT1, OUT1);
   check("OUT2", old_OUT2, OUT2);
   check("OUT3", old_OUT3, OUT3);

   // Alle Client-Werte prüfen
   check("c0out1", old_c0out1, c0out1);
   check("c0out2", old_c0out2, c0out2);
   check("c0out3", old_c0out3, c0out3);

   check("c1out1", old_c1out1, c1out1);
   check("c1out2", old_c1out2, c1out2);
   check("c1out3", old_c1out3, c1out3);

   check("c2out1", old_c2out1, c2out1);
   check("c2out2", old_c2out2, c2out2);
   check("c2out3", old_c2out3, c2out3);

   check("c3out1", old_c3out1, c3out1);
   check("c3out2", old_c3out2, c3out2);
   check("c3out3", old_c3out3, c3out3);

   // ----------------------------------------------------
   // CHANGE LOGGER – Event-Erkennung
   // ----------------------------------------------------
   // Wichtig: NICHT next_WebserverMessage hier auf "" setzen (würde Signal löschen).
   // next_WebserverMessage wird nur hier gesetzt auf "!!!NEXT" beim ersten Erkennen.

   if (changed) {
      // mark warning active (dauerhaft until user/logic clears)
      changeWarningActive = true;

      // Nur beim ersten Erkennen des Events fixieren (einmalig pro Event)
      if (!sav_timestr_set_for_event) {
         // Zeitpunkt einmalig festhalten
         sav_timestr = timestr;

         // Message-Snapshots:
         // - prev = die Message vor der affiliat (vor-vor)
         // - affiliat = die Message, die zum Zeitpunkt DES EVENTS aktuell war
         //   (also die 'letzte gesendete' vor diesem Erkennungsdurchlauf)
         prev_WebserverMessage = savold_WebserverMessage;   // vorherige-vorherige
         // old_ wurde bereits vor Aufruf auf aktuelle auslösende msg gesetzt
         affiliat_ChangeMessage = old_WebserverMessage;
         // Setze Signal, damit handleClients() die nächste gesendete Message als 'after' erfassen kann
         next_WebserverMessage = "!!!NEXT";

         // Flag, dass die event-snapshots für dieses Event gesetzt wurden
         sav_timestr_set_for_event = true;
      }

      // Debug-Log (zeigt die eingefrorenen Snapshots)
      Serial.println("[CHANGE] Previous Message (before affiliated):");
      Serial.println("   " + prev_WebserverMessage);
      Serial.println("[CHANGE] Affiliated Message (triggering event):");
      Serial.println("   " + affiliat_ChangeMessage);
   }

   // Hinweis:
   // sav_timestr_set_for_event wird in handleClients() wieder freigegeben,
   // sobald die erste gültige nach-affiliated Message erfasst wurde.
   // changeWarningActive bleibt true bis extern zurückgesetzt (oder durch Logik).

} // Ende: ChangeLogger()
